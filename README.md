let sortedAlphabetically = false;

function plainName(s) {
  return (s || "").replace(/\s*\([^)]*\)/g, "").trim();
}
function normalize(s) {
  return (s || "").toLocaleLowerCase("ru-RU")
    .replace(/ё/g, "е")
    .replace(/[^\p{L}\p{N}\s-]/gu, " ")
    .replace(/\s+/g, " ").trim();
}
function commonsSearchUrl(query) {
  return "https://commons.wikimedia.org/wiki/Special:MediaSearch?type=image&search=" +
    encodeURIComponent(query);
}
function setPhotoStatus(status, message, fileUrl) {
  status.replaceChildren();
  status.append(document.createTextNode(message));
  if (fileUrl) {
    status.append(document.createTextNode(" "));
    const source = document.createElement("a");
    source.className = "source-link";
    source.href = fileUrl;
    source.target = "_blank";
    source.rel = "noopener noreferrer";
    source.textContent = "Страница источника";
    status.append(source);
  }
}
function makeCard(hero, index) {
  const card = document.createElement("article");
  card.className = "hero-card";
  card.dataset.search = normalize(hero.name + " " + hero.bio);
  card.dataset.name = normalize(plainName(hero.name));

  const wrap = document.createElement("div");
  wrap.className = "portrait-wrap";

  const img = document.createElement("img");
  img.className = "portrait";
  img.alt = "Возможное архивное изображение для биографии: " + hero.name;
  img.loading = "lazy";
  img.decoding = "async";
  img.referrerPolicy = "no-referrer";
  img.hidden = true;

  const placeholder = document.createElement("div");
  placeholder.className = "photo-placeholder";
  placeholder.setAttribute("aria-hidden", "true");
  placeholder.textContent = "Поиск архивного изображения…";
  placeholder.style.cssText =
    "padding:20px;text-align:center;color:#687386;font-weight:700;max-width:260px";

  const note = document.createElement("div");
  note.className = "photo-note";
  note.textContent = "Ищем изображение в открытых архивах";
  wrap.append(img, placeholder, note);

  const body = document.createElement("div");
  body.className = "hero-body";

  const num = document.createElement("span");
  num.className = "number";
  num.textContent = "Биография № " + (index + 1);

  const title = document.createElement("h2");
  title.textContent = hero.name;

  const bio = document.createElement("p");
  bio.textContent = hero.bio;

  const status = document.createElement("p");
  status.className = "photo-status";
  status.textContent = "Подготавливаем поиск изображения…";

  const link = document.createElement("a");
  link.className = "photo-link";
  link.target = "_blank";
  link.rel = "noopener noreferrer";
  link.href = commonsSearchUrl(plainName(hero.name));
  link.textContent = "Проверить архивные изображения";

  body.append(num, title, bio, status, link);
  card.append(wrap, body);
  grid.append(card);

  return { hero, card, img, placeholder, note, status, link };
}

const cards = heroes.map(makeCard);

function updateFilter() {
  const q = normalize(search.value);
  let visible = 0;

  for (const item of cards) {
    const matches = !q || item.card.dataset.search.includes(q);
    item.card.hidden = !matches;
    if (matches) visible++;
  }

  count.textContent = q
    ? `${visible} из ${heroes.length} биографий`
    : `${heroes.length} биографии`;

  resultsSummary.textContent = q
    ? `По запросу «${search.value.trim()}» найдено: ${visible}. Поиск учитывает имя и текст биографии.`
    : `В каталоге ${heroes.length} биографии. Фотографии загружаются из открытых источников и требуют проверки.`;

  emptyState.classList.toggle("visible", visible === 0);
}

search.addEventListener("input", updateFilter);

clearSearch.addEventListener("click", () => {
  search.value = "";
  updateFilter();
  search.focus();
});

cards.forEach((item, index) => {
  item.originalIndex = index;
});

sortButton.addEventListener("click", () => {
  sortedAlphabetically = !sortedAlphabetically;

  const ordered = [...cards].sort((a, b) => {
    const cmp = a.hero.name.localeCompare(b.hero.name, "ru");
    return sortedAlphabetically ? cmp : a.originalIndex - b.originalIndex;
  });

  ordered.forEach(item => grid.append(item.card));
  sortButton.setAttribute("aria-pressed", String(sortedAlphabetically));
  sortButton.textContent = sortedAlphabetically
    ? "Вернуть исходный порядок"
    : "Сортировать по алфавиту";
});

updateFilter();

async function worker() {
  while (queue.length) {
    const item = queue.shift();
    if (item) await findPortrait(item);
  }

async function apiJson(url, timeoutMs = 9000) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), timeoutMs);

  try {
    const response = await fetch(url, {
      signal: controller.signal,
      headers: { Accept: "application/json" }
    });

    if (!response.ok) throw new Error("HTTP " + response.status);
    return await response.json();
  } finally {
    clearTimeout(timeout);
  }
}

function buildApiUrl(base, params) {
  return base + "?" +
    new URLSearchParams({ ...params, origin: "*" }).toString();
}

function setImage(item, src, label, message, sourceUrl) {
  return new Promise(resolve => {
    let settled = false;

    const done = value => {
      if (!settled) {
        settled = true;
        resolve(value);
      }
    };

    item.img.onload = () => {
      item.img.hidden = false;
      item.placeholder.hidden = true;
      item.note.textContent = label;
      setPhotoStatus(item.status, message, sourceUrl);
      done(true);
    };

    item.img.onerror = () => {
      item.img.removeAttribute("src");
      item.img.hidden = true;
      item.placeholder.hidden = false;
      done(false);
    };

    item.img.src = src;
  });
}

function matchesPerson(title, person) {
  const t = normalize(
    title.replace(/^File:/i, "").replace(/\.[a-z0-9]{2,5}$/i, "")
  );
  const tokens = normalize(person).split(" ").filter(Boolean);
  return tokens.length > 0 &&
    tokens.slice(0, 2).every(token => t.includes(token));
}

async function findWikipediaImage(item, person) {
  const base = "https://ru.wikipedia.org/w/api.php";

  const params = {
    action: "query",
    format: "json",
    formatversion: "2",
    list: "search",
    srsearch: person,
    srnamespace: "0",
    srlimit: "8"
  };

  const data = await apiJson(buildApiUrl(base, params));
  const pages = data.query?.search || [];

  const ordered = [...pages].sort((a, b) =>
    Number(matchesPerson(b.title, person)) -
    Number(matchesPerson(a.title, person))
  );

  for (const found of ordered) {
    const imageParams = {
      action: "query",
      format: "json",
      formatversion: "2",
      prop: "pageimages",
      piprop: "thumbnail",
      pithumbsize: "700",
      titles: found.title,
      redirects: "1"
    };

    const imageData = await apiJson(buildApiUrl(base, imageParams));
    const page = (imageData.query?.pages || [])
      .find(p => p.thumbnail?.source);

    if (!page) continue;

    const exactish = matchesPerson(page.title, person);
    const sourceUrl = "https://ru.wikipedia.org/wiki/" +
      encodeURIComponent(page.title.replace(/ /g, "_"));

    const ok = await setImage(
      item,
      page.thumbnail.source,
      exactish
        ? "Изображение из русской Википедии"
        : "Изображение из найденной статьи",
      exactish
        ? "Источник: статья «" + page.title +
          "». Совпадение по имени не подтверждает личность на снимке."
        : "Источник: статья «" + page.title +
          "». Это косвенное совпадение; фото может быть не связано с героем.",
      sourceUrl
    );

    if (ok) return true;
  }

  return false;
}

async function findCommonsImage(item, query, person) {
  const base = "https://commons.wikimedia.org/w/api.php";

  const params = {
    action: "query",
    generator: "search",
    gsrnamespace: "6",
    gsrsearch: query,
    gsrlimit: "20",
    prop: "imageinfo",
    iiprop: "url",
    iiurlwidth: "700"
  };

  const data = await apiJson(buildApiUrl(base, params));

  const pages = (data.query?.pages || []).filter(p =>
    p.imageinfo?.[0] &&
    (p.imageinfo[0].thumburl || p.imageinfo[0].url)
  );

  if (!pages.length) return false;

  pages.sort((a, b) =>
    Number(matchesPerson(b.title, person)) -
    Number(matchesPerson(a.title, person))
  );

  for (const page of pages) {
    const info = page.imageinfo[0];
    const exactish = matchesPerson(page.title, person);

    const fileUrl = "https://commons.wikimedia.org/wiki/" +
      encodeURIComponent(page.title.replace(/ /g, "_"));

    const ok = await setImage(
      item,
      info.thumburl || info.url,
      exactish
        ? "Возможное фото по совпадению имени"
        : "Архивное изображение по близкому запросу",
      "Файл: «" + page.title.replace(/^File:/, "") + "». " +
        (exactish
          ? "Совпадение названия не гарантирует, что изображён именно этот человек. Проверьте подпись и описание файла."
          : "Неточное совпадение: изображение может не иметь отношения к герою. Проверьте описание файла."),
      fileUrl
    );

    if (ok) return true;
  }

  return false;
}

async function findPortrait(item) {
  const person = plainName(item.hero.name);
  const parts = person.split(/\s+/);
  const surname = parts[0] || "";
  const first = parts[1] || "";

  const queries = [
    person,
    surname + " " + first + " орден Славы",
    surname + " орден Славы",
    person + " Великая Отечественная война",
    "полные кавалеры ордена Славы Саратовская область",
    "советские солдаты Великая Отечественная война архивная фотография"
  ];

  item.link.href = commonsSearchUrl(person);
  item.note.textContent = "Ищем в Википедии и архивах…";

  try {
    if (await findWikipediaImage(item, person)) return;
  } catch (error) {
    // Если Википедия недоступна, продолжаем поиск в Commons.
  }

  for (const query of queries) {
    try {
      if (await findCommonsImage(item, query, person)) return;
    } catch (error) {
      // Переходим к следующему запросу.
    }
  }

  item.img.hidden = true;
  item.placeholder.hidden = false;
  item.placeholder.textContent = "Изображение не найдено автоматически";
  item.note.textContent = "Откройте архивный поиск";

  setPhotoStatus(
    item.status,
    "Автоматический поиск не нашёл доступного изображения. Проверьте результаты вручную по ссылке ниже."
  );
}
