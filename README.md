const grid = document.getElementById("heroes");
const search = document.getElementById("search");
const count = document.getElementById("count");

function plainName(s) {
  return s.replace(/\s*\([^)]*\)/g, "").trim();
}

function makeCard(hero, index) {
  const card = document.createElement("article");
  card.className = "hero-card";
  card.dataset.search = (hero.name + " " + hero.bio).toLocaleLowerCase("ru");

  const wrap = document.createElement("div");
  wrap.className = "portrait-wrap";

  const img = document.createElement("img");
  img.className = "portrait";
  img.alt = "Портрет: " + hero.name;
  img.loading = "lazy";

  const note = document.createElement("div");
  note.className = "photo-note";
  note.textContent = "Ищем архивный портрет";
  wrap.append(img, note);

  const body = document.createElement("div");
  body.className = "hero-body";

  const num = document.createElement("span");
  num.className = "number";
  num.textContent = "Герой № " + (index + 1);

  const title = document.createElement("h2");
  title.textContent = hero.name;

  const bio = document.createElement("p");
  bio.textContent = hero.bio;

  const status = document.createElement("p");
  status.className = "photo-status";
  status.textContent = "Поиск изображения в Wikimedia Commons…";

  const link = document.createElement("a");
  link.className = "photo-link";
  link.target = "_blank";
  link.rel = "noopener noreferrer";
  link.href = "https://commons.wikimedia.org/wiki/Special:MediaSearch?type=image&search=" +
    encodeURIComponent(plainName(hero.name));
  link.textContent = "Проверить архивные изображения";

  body.append(num, title, bio, status, link);
  card.append(wrap, body);
  grid.append(card);

  findPortrait(img, hero.name, note, status, wrap);
}

async function findPortrait(img, fullName, note, status, wrap) {
  const person = plainName(fullName);
  const parts = person.split(/\s+/);
  const surname = parts[0] || "";
  const first = parts[1] || "";

  const queries = [
    '"' + person + '"',
    person + " портрет",
    surname + " " + first + " орден Славы"
  ];

  for (const q of queries) {
    try {
      const url =
        "https://commons.wikimedia.org/w/api.php?action=query" +
        "&generator=search&gsrnamespace=6&gsrlimit=20" +
        "&gsrsearch=" + encodeURIComponent(q) +
        "&prop=imageinfo&iiprop=url&iiurlwidth=600" +
        "&format=json&origin=*";

      const response = await fetch(url);
      if (!response.ok) continue;

      const data = await response.json();
      const pages = Object.values(data.query?.pages || {});

      const norm = s =>
        (s || "").toLocaleLowerCase("ru").replace(/ё/g, "е");

      const match = pages
        .map(p => ({ page: p, title: norm(p.title) }))
        .find(x =>
          x.page.imageinfo?.[0] &&
          x.title.includes(norm(surname)) &&
          x.title.includes(norm(first))
        );

      if (match) {
        img.src = match.page.imageinfo[0].thumburl ||
                  match.page.imageinfo[0].url;

        note.textContent = "Источник изображения: Wikimedia Commons";
        status.textContent =
          "Проверьте подпись и описание: совпадение по имени не гарантирует личность на портрете.";

        img.onerror = () => {
          img.remove();
          note.textContent = "Портрет не загрузился";
          status.textContent = "Портрет можно поискать по ссылке ниже.";
        };
        return;
      }
    } catch (error) {
      // Если поиск не сработал, карточка биографии остаётся на странице.
    }
  }

  img.remove();
  note.textContent = "Портрет не найден автоматически";
  status.textContent =
    "Надёжное совпадение не найдено. Откройте поиск портрета по ссылке ниже.";
}

heroes.forEach(makeCard);

search.addEventListener("input", () => {
  const q = search.value.trim().toLocaleLowerCase("ru");
  let visible = 0;

  document.querySelectorAll(".hero-card").forEach(card => {
    const matches = card.dataset.search.includes(q);
    card.hidden = !matches;
    if (matches) visible++;
  });

  count.textContent = visible + " из " + heroes.length + " биографий";
});
</script>
